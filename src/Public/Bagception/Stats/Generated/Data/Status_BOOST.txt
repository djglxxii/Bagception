// BAGCEPTION_WEIGHTLESS makes Bagception's contents weigh nothing. The
// Bagception_Probe goal applies it to every item that goes into a Bagception or one of
// its bags, and removes it when the item leaves; a sweep on every load repairs
// anything missed. Weight() is additive and clamps at zero per item, so one boost
// larger than any item zeroes it without the mod needing to know what it weighed.
// See the decision log, 2026-09-20 to 2026-09-23.

new entry "BAGCEPTION_WEIGHTLESS"
type "StatusData"
data "StatusType" "BOOST"
data "DisplayName" "he9d45eb5gc833g41e1gbfc4gfc5c153c29bd"
data "Description" "h96dac754gd7e3g46begacb7gb707b14b3374"
data "StackId" "BAGCEPTION_WEIGHTLESS"
data "StatusPropertyFlags" "IgnoreResting;IgnoredByImmobilized;DisableOverhead;DisableCombatlog;ApplyToDead;DisablePortraitIndicator"
data "Boosts" "Weight(-100000)"

// RETIRED: the carry boost

// master bag. Weightless contents replaced it in 1.0.0.5. The master bag no longer
